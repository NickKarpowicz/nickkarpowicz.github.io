55    """Calculates the transform-limited pulse corresponding to a spectrum
56
57    Args:
58        wavelength_nm: the wavelengths included in the data, in nanometers
59        spectrum: the spectrum corresponding to the input wavelength scale
60        gate_level: (optional) level, relative to the maximum at which to apply a gate to the spectrum. For example, with gate_level=0.01, values less than 1% of the maximum signal will be set to zero
61
62    Returns:
63        t: time vector (s)
64        pulse: the pulse intensity vs. time
65    """
66    f,spec = wavelength_to_frequency(wavelengths_nm, spectrum)
67    df = f[1]-f[0]
68    t = np.fft.fftshift(np.fft.fftfreq(f.shape[0],d=df))

71        gated_spectrum[spec<(gate_level * np.max(spec))] = 0.0
72
73    pulse = np.fft.fftshift(
74        np.abs(
75            np.fft.ifft(
76                np.sqrt(gated_spectrum))))**2

382    def compute_calibration_factor_spectrometer(self, transmission_additional_optics=None, smoothing='poly', extend_calibration: bool = False, wavelength_ROI: list = [420, 800]):
383        """Computes the calibration factor.
384
385         It assumes that in the SpectrumHandler object:
386         * the measured spectrum of the calibration lamp is loaded as main spectrum
387         * the true tabulated spectrum of the calibration lamp is loaded as self.calibration_lamp_spectrum [via load_calibration_lamp_data()]
388
389         Args:
390            transmission_additional_optics: list of SpectrumHandler objects = list of transmission spectra of additional optics to be taken into account when computing the calibration factor.
391                The transmission ['reflection' for mirrors] spectra of additional optics will be included (multiplied) in the calibration factor
392                Additional optics are assumed to be in the beam path of any standard measurement (e.g. integrating sphere) BUT NOT PRESENT during the calibration lamp measurement.
393            smoothing: Type of smoothing to be applied to the calibration factor. Options are 'poly' (default) None or an integer number. If 'poly', a polynomial fit is applied to the calibration factor.
394                If None, no smoothing is applied. If an integer number, a box filter of that size is applied to the calibration factor.
395            extend_calibration (bool): If True, the calibration factor is extended to the full wavelength range of the spectrometer (the first and last values of the calibration curve are used for the extension).
396                This is a quick and dirty solution to deal with the fact that the calibration lamp spectrum is not available for the full wavelength range of the spectrometer.
397            wavelength_ROI (list): [wvl1, wvl2] Wavelength range for the normalization of the calibration factor (for display purposes).
398                Default is [420, 800] nm."""
399        if self.calibration_lamp_wvl is None or self.calibration_lamp_spectrum is None:
400            raise ValueError("Calibration lamp data not loaded.")
401        wvl_stored, spectrum_stored = self.wvl, self.spectrum
402        self.divide_by('calibration data')
403
404        eliminate_outliers(self.spectrum, threshold=3, window_points=20)
405
406        if transmission_additional_optics is not None:
407            for sao in transmission_additional_optics:
408                if not isinstance(sao, SpectrumHandler):
409                    raise TypeError("Input of spectra_additional_optics must be a list of SpectrumHandler objects.")
410                wvl_sao, spectrum_sao = sao.get_spectrum()
411                if np.any(np.diff(wvl_sao)) < 0:
412                    raise ValueError("Wavelength array of additional optics spectrum must be sorted in ascending order.")
413                wvl_sao = np.concatenate((np.array([0]), wvl_sao, np.array([max(np.max(self.wvl), np.max(wvl_sao))])))
414                spectrum_sao = np.concatenate((np.array([spectrum_sao[0]]), spectrum_sao, np.array([spectrum_sao[-1]])))
415                self.spectrum = self.spectrum * np.interp(self.wvl, wvl_sao, spectrum_sao)
416
417        self.calibration_factor = self.spectrum
418        if smoothing is None:
419            pass
420        elif smoothing == 'poly':
421            smoothed_cf = np.poly1d(np.polyfit(self.wvl, self.calibration_factor, 50))
422            self.calibration_factor = smoothed_cf(self.wvl)
423        else:
424            if not isinstance(smoothing, int):
425                raise TypeError("Input of smoothing must be 'poly', None or an integer.")
426            if smoothing < 1:
427                raise ValueError("Input of smoothing must be a positive integer.")
428            self.calibration_factor = box_smooth(self.calibration_factor, smoothing)
429
430        for i in range(len(self.wvl)):
431            if self.calibration_factor[i] < 0.01:
432                self.calibration_factor[i] = 0.01
433        self.calibration_factor = 1/self.calibration_factor
434        self.calibration_factor = self.calibration_factor / np.mean(self.calibration_factor[
435            (self.wvl > wavelength_ROI[0]) & (self.wvl < wavelength_ROI[1])])
436
437        if extend_calibration:
438            # Extend the calibration factor to the full wavelength range of the spectrometer (pick the first and last values of the calibration curve for the extension)
439            self.wvl = np.concatenate((np.linspace(0, self.wvl[0], 100), self.wvl, np.linspace(self.wvl[-1], self.wvl[-1], 100)))
440            self.calibration_factor = np.concatenate((np.ones(100)*self.calibration_factor[0], self.calibration_factor, np.ones(100)*self.calibration_factor[-1]))
441            self.calibration_factor = np.interp(wvl_stored, self.wvl, self.calibration_factor)
442            self.wvl = wvl_stored
443            self.spectrum = spectrum_stored
444        else:
445            self.spectrum = spectrum_stored[(wvl_stored>np.min(self.calibration_lamp_wvl))&(wvl_stored<np.max(self.calibration_lamp_wvl))]
