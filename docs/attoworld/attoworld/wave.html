259    def load_trace_from_spectral_data(self, wvl_FFT_trace, spectrum_FFT_trace, phase_FFT_trace):
260        """loads the trace from wavelength, spectrum, and spectral phase arrays and stores it in the class.
261
262        This function has a different job than load_spectrum_from_arrays(): in fact, it retrieves the trace from the given spectral data via fourier transform.
263
264        ARGUMENTS:
265            wvl_FFT_trace: wavelength array in nm
266            spectrum_FFT_trace: (wavelength-)spectral intensity array. This is assumed to be |FFT{field}|^2 * df/dλ
267            phase_FFT_trace: spectral phase in rad
268        """
269        check_equal_length(wvl_FFT_trace, spectrum_FFT_trace, phase_FFT_trace)
270
271        # check that the wvl_FFT_trace array is monotonically increasing
272        if np.any(np.diff(wvl_FFT_trace) < 0):
273            if np.all(np.diff(wvl_FFT_trace) < 0):
274                wvl_FFT_trace = wvl_FFT_trace[::-1]
275                spectrum_FFT_trace = spectrum_FFT_trace[::-1]
276                phase_FFT_trace = phase_FFT_trace[::-1]
277            else:
278                raise ValueError('in function TraceHandler.load_from_spectral_data() wavelength array is not monotonous')
279
280        # frequency spectrum (monotonically increasing)
281        freq = c / wvl_FFT_trace[::-1] * 1e-6
282        spectrum_freq = spectrum_FFT_trace[::-1] / (c/wvl_FFT_trace[::-1]**2 *1e-6)
283        phase_FFT_trace = phase_FFT_trace[::-1]
284
285        # create proper freq axis for the fourier transform
286        df = (freq[1]-freq[0])/4
287        nf = int(np.ceil(3 * freq[-1]/df))

289        # corresponding time window

359    def update_spectral_phase(self):
360        """computes the spectral phase of the trace. Result is stored in self.fftphase.
361
362        An attempt of automatic linear phase subtraction is made. Phase is unwrapped.
363        """
364        # determine the time distance (delay) of the waveform peak from the beginning of the trace (more precisely, from the beginning of the ifft)
365        tmax = self.get_zero_delay()
366        deltaT1 = tmax - self.complexFieldTimeV[0]
367        deltaT2 = self.complexFieldTimeV[-1] - tmax
368        ifft_twindow = 1/ (self.frequencyAxis[1] - self.frequencyAxis[0])
369        total_delay = ifft_twindow/2 + (deltaT2-deltaT1)/2 # total estimated delay of the peak from the beginning of the ifft trace
370
371        # compute the phase, use the computed 'delay' to correct for the non-interesting linear component, and unwrap it
372        self.fftphase = np.angle(self.fftFieldV[self.frequencyAxis > 0])
373        self.fftphase = self.fftphase - total_delay * self.frequencyAxis[self.frequencyAxis > 0] * 2 * np.pi
374        self.fftphase = np.unwrap(self.fftphase)
375
376        # subtract constant phase offset (m * 2*π)
377        n_pi =  np.sum(self.fftSpectrum*self.fftphase)/ np.sum(self.fftSpectrum)
378        n_pi = int( n_pi / ( 2*np.pi ))
379        self.fftphase -= n_pi * 2 * np.pi

571        """returns the positive frequency axis f and the corresponding FFT field (complex).
572
573        Main task of this function is to 'smooth' the phase of the fft (= remove linear phase, corresponding to a time shift in the temporal domain), since the waveform is usually centered around t = 0"""
574        if self.frequencyAxis is None or self.fftFieldV is None:
575            raise ValueError('in function TraceHandler.get_positive_fft_field() frequency axis or fft field is not defined')
576        positive_freq = self.frequencyAxis[self.frequencyAxis >= 0]

578        # remove linear phase
579        tmax = self.get_zero_delay()

582        ifft_twindow = 1 / (self.frequencyAxis[1] - self.frequencyAxis[0])

670    def fft_tukey_bandpass(self, lowWavelengthEdge, upWavelengthEdge, lowEdgeWidth, highEdgeWidth):
671        """applies a bandpass filter to the trace in the frequency domain using a tukey window.
672
673        The tukey window is 1 between  lowWavelengthEdge+lowEdgeWidth/2  and  upWavelengthEdge-highEdgeWidth/2
674        and it is reaches zero at lowWavelengthEdge-lowEdgeWidth/2  and  upWavelengthEdge+highEdgeWidth/2.
675        Notice that the edges are only cosine-shaped in the frequency domain. In the wavelength domain upWavelengthEdge - lowWavelengthEdge does not coincide with the FWHM of the tukey function
676
677        ARGUMENTS:
678            lowWavelengthEdge: float (nm)
679            upWavelengthEdge: float (nm)
680            lowEdgeWidth: float (nm)
681            highEdgeWidth: float (nm)
682        """
683        upFreqEdge = c / (lowWavelengthEdge - lowEdgeWidth / 2) /2 * 1e-6 + c / (
684                    lowWavelengthEdge + lowEdgeWidth / 2) /2 * 1e-6
685        lowFreqEdge = c / (upWavelengthEdge - highEdgeWidth / 2) /2 * 1e-6 + c / (
686                    upWavelengthEdge + highEdgeWidth / 2) /2 * 1e-6
687
688        upFreqEdgeWidth = c / (lowWavelengthEdge - lowEdgeWidth / 2) * 1e-6 - c / (
689                    lowWavelengthEdge + lowEdgeWidth / 2) * 1e-6
690        lowFreqEdgeWidth = c / (upWavelengthEdge - highEdgeWidth / 2) * 1e-6 - c / (
691                    upWavelengthEdge + highEdgeWidth / 2) * 1e-6
692
693        window = asymmetric_tukey_window(np.abs(self.frequencyAxis), lowFreqEdge, upFreqEdge, lowFreqEdgeWidth,
694                                         upFreqEdgeWidth)
695        self.fftFieldV = self.fftFieldV * window
696        self.update_trace_from_fft()
697        self.update_fft_spectrum()
698        self.strip_from_trace()

786    def fresnel_reflection(self, material2, angle_in, material1=None, forward: bool=True, s_polarized: bool=True, path: str = "./RefractiveIndices/"):
787        """calculates the fresnel reflection of the pulse at the interface between two materials. The waveform is travelling from material 1 to material 2.
788        The first medium (material1) should be non-absorptive. As usual the resulting waveform is stored in the TraceHandler object, replacing the previous one.
789

792            material2 = the filename (without '.txt') of the refractive index data for the material after the interface (e.g. Si Al MgF2); wavelength is in um in the refractive index file
793            angle_in = the incidence angle in degrees
794
795        OPTIONAL ARGUMENTS:
796            material1 = the filename (without '.txt') of the refractive index data for the material before the interface. If None (default), vacuum is assumed
797            forward: bool = if True (default) forward reflection is computed (the result waveform is the reflection of the previously stored waveform.
798                if False backward reflection is computed (the previous waveform is the reflection of the result waveform)

803            raise ValueError('in function TraceHandler.fresnel_reflection() p_polarized is not implemented yet\n')

807        wvl2 = np.array(refIndexData['wvl']) * 1e3
808        n2 = np.array(refIndexData['n']) +1j* np.array(refIndexData['k'])
809        if material1 is None:
810            wvl1 = np.array(wvl2)
811            n1 = n2*0 + 1
812        else:
813            refIndexData = pandas.read_table(path + material1 + ".txt", sep=" ", keep_default_na=False)

815            n1 = np.array(refIndexData['n']) +1j* np.array(refIndexData['k'])

818        if np.any(np.diff(wvl1) < 0):
819            if np.all(np.diff(wvl1) < 0):
820                wvl1 = wvl1[::-1]
821                n1 = n1[::-1]
822            else:
823                raise ValueError('in function TraceHandler.fresnel_reflection() wavelength array is not monotonous')
824        if np.any(np.diff(wvl2) < 0):
825            if np.all(np.diff(wvl2) < 0):
826                wvl2 = wvl2[::-1]
827                n2 = n2[::-1]
828            else:
829                raise ValueError('in function TraceHandler.fresnel_reflection() wavelength array is not monotonous')
830
831        # frequency spectrum (monotonically increasing)
832        freq1 = c / wvl1[::-1] * 1e-6
833        freq2 = c / wvl2[::-1] * 1e-6
834        n1 = n1[::-1]
835        n2 = n2[::-1]
836
837        # fill with ones
838        initial_ones_freq = np.linspace(freq1[1] - freq1[0], freq1[0], int(np.ceil(4 * freq1[0] / (freq1[1] - freq1[0]))))

849                                        int(np.ceil(15 * freq2[-1] / (freq2[-1] - freq2[-2]))))

851        freq2 = np.concatenate((initial_ones_freq, freq2, final_ones_freq))

854        # add negative frequencies
855        freq1 = np.concatenate((-freq1[::-1], freq1))
856        n1 = np.concatenate((np.conjugate(n1[::-1]), n1))
