39def load_waves_from_matfile(filename: str, phase: Optional[float] = None):
40    """Load the contents of an attolab scanner file in .mat format
41
42    Args:
43        phase (float): phase to use when interpreting the lock-in data
44        filename (str): path to the mat file
45    Returns:
46        time_delay: array of time delay values
47        signal: signals corresponding to the time delays
48    """
49
50    datablob = sio.loadmat(filename)
51    stage_position = datablob['xdata'][0,:]
52    time_delay = -2e-3 * stage_position/2.9979e8
53    lia_x = datablob['x0']
54    lia_y = datablob['y0']
55    if phase is None:
56        optimized_phase = np.atan2(np.sum(lia_y[:]**2), np.sum(lia_x[:]**2))
57        signal = np.fliplr(lia_x*np.cos(optimized_phase) + lia_y*np.sin(optimized_phase))
58        return time_delay, signal
59    else:
60        signal = np.fliplr(lia_x*np.cos(phase) - lia_y*np.sin(phase))
61        return time_delay, signal

17    """Loads a waveform from a text file
18
19    Args:
20        filename (str): path to the file
21        time_multiplier (float): multiplier needed to convert the time unit in the file to seconds
22        time_field (str): name (header) of the column containing the times
23        wave_field (str): name (header) of the column containing the waveform
24        sep (str): separator used in the file format (tab is default)
25
26    Returns:
27        Waveform: the waveform
28    """
29
30    data = pandas.read_csv(filename, sep=sep)
31    time = time_multiplier * data[time_field].to_numpy()
32    wave = data[wave_field].to_numpy()
33    dt = time[1]-time[0]
34    diff_time = np.diff(time)
35    uniform = bool(np.all(np.isclose(diff_time, diff_time[0])))
36    return Waveform(wave = wave, time = time, dt=dt, is_uniformly_spaced = uniform)
