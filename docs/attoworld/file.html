28    """
29    with open(file_path, "r") as file:
30        lines = file.readlines()
31        delay_increment = float(lines[2].strip().split("=")[1])
32
33    wavelength_vector = pd.read_csv(
34        file_path, skiprows=5, nrows=1, delimiter="\t", header=None

41    freqs, first_spec = wavelength_to_frequency(wavelength_vector, data_array[:, 0])
42    if freqs is not None:
43        data_array_freq = np.zeros((len(freqs), data_array.shape[1]))
44        data_array_freq[:, 0] = first_spec
45        for _i in range(data_array.shape[1]):
46            _f, _dat = wavelength_to_frequency(
47                wavelength_vector, data_array[:, _i], freqs
48            )
49            data_array_freq[:, _i] = _dat

127
128    datablob = sio.loadmat(filename)

893    def get_error(self) -> float:
894        """
895        Get the G' error of the reconstruction
896        """
897        norm_measured = np.linalg.norm(self.measured_spectrogram.data)
898        norm_retrieved = np.linalg.norm(self.reconstructed_spectrogram.data)
899        return np.sqrt(
900            np.sum(
901                (
902                    self.measured_spectrogram.data[:] / norm_measured
903                    - self.reconstructed_spectrogram.data[:] / norm_retrieved
904                )
905                ** 2
906            )
907            / np.sum((self.measured_spectrogram.data[:] / norm_measured) ** 2)
908        )

910    def get_G_error(self) -> float:
911        """

914        """
915        return np.sqrt(
916            (1.0 / float(len(self.measured_spectrogram.data[:]) ** 2))

918                (
919                    self.measured_spectrogram.data[:]

921                    - self.reconstructed_spectrogram.data[:]
922                    / np.max(self.reconstructed_spectrogram.data[:])
923                )
924                ** 2
925            )
926        )

105    ):
106        """Bin a spectrogram to a FFT-appropriate shape
107

123                _f, self.freq, np.array(self.data[:, _i]), neighbors=2

132                _t, self.time - t0, np.array(binned_data[_i, :]), neighbors=2
133            )
