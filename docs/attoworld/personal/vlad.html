  33def soft_window(x_grid: ArrayLike,
  34                x_begin: float,
  35                x_end: float) -> ArrayLike:
  36  """Computes a soft window function.
  37
  38  The window smoothly transitions from 1 to 0 over the interval
  39  [min(x_begin, x_end), max(x_begin, x_end)].
  40  If x_begin <= x_end, it's a cosine-squared fall-off.
  41  If x_begin > x_end, it's a sine-squared rise-on.
  42
  43  Args:
  44    x_grid: A 1D array of x-coordinates.
  45    x_begin: The x-coordinate where the transition starts (value is 1).
  46    x_end: The x-coordinate where the transition ends (value is 0).
  47
  48  Returns:
  49    A 1D array of the same size as x_grid, containing the window values.
  50  """
  51  window = np.zeros_like(x_grid)
  52  x_min_transition = min(x_begin, x_end)
  53  x_max_transition = max(x_begin, x_end)
  54
  55  # Determine indices for different parts of the window
  56  indices_before_transition = np.where(x_grid < x_min_transition)[0]
  57  idx_transition_start = (indices_before_transition[-1] + 1
  58                          if indices_before_transition.size > 0 else 0)
  59
  60  indices_after_transition = np.where(x_grid > x_max_transition)[0]
  61  idx_transition_end = (indices_after_transition[0]
  62                        if indices_after_transition.size > 0
  63                        else len(x_grid))
  64
  65  # Define the transition region
  66  x_transition = x_grid[idx_transition_start:idx_transition_end]
  67
  68  if x_begin <= x_end: # Window goes from 1 down to 0
  69    window[:idx_transition_start] = 1.0
  70    if idx_transition_end > idx_transition_start and x_max_transition > x_min_transition:
  71      window[idx_transition_start:idx_transition_end] = \
  72          np.cos(np.pi / 2.0 * (x_transition - x_min_transition) /
  73                 (x_max_transition - x_min_transition))**2
  74    # Values after x_end remain 0 (initialized)
  75  else: # Window goes from 0 up to 1 (x_begin > x_end, so x_min_transition = x_end)
  76    window[idx_transition_end:] = 1.0
  77    if idx_transition_end > idx_transition_start and x_max_transition > x_min_transition:
  78      window[idx_transition_start:idx_transition_end] = \
  79          np.sin(np.pi / 2.0 * (x_transition - x_min_transition) /
  80                 (x_max_transition - x_min_transition))**2
  81    # Values before x_begin (which is x_max_transition) remain 0 (initialized)
  82  return window

  85def get_significant_part_indices_v1(
  86    array_data: ArrayLike,
  87    threshold: float = 1e-8) -> Tuple[int, int]:
  88  """Returns indices (i1, i2) for the slice A[i1:i2] containing the
  89  significant part of the array.
  90
  91  The significant part is defined relative to the maximum absolute value
  92  in the array. Elements A[:i1] and A[i2:] are considered "small".
  93
  94  Args:
  95    array_data: The input 1D array.
  96    threshold: The relative threshold to determine significance.
  97               Elements are significant if abs(element) >= threshold * max(abs(array_data)).
  98
  99  Returns:
 100    A tuple (i1, i2) representing the start (inclusive) and
 101    end (exclusive) indices of the significant part.
 102  """
 103  abs_array = np.abs(array_data)
 104  if abs_array.size == 0:
 105    return 0, 0
 106  
 107  idx_max = np.argmax(abs_array)
 108  array_max_val = abs_array[idx_max]
 109
 110  if array_max_val == 0: # All elements are zero
 111      return 0, len(array_data)
 112
 113  significant_indices_before_max = np.where(
 114      abs_array[:idx_max] >= threshold * array_max_val)[0]
 115  i1 = (significant_indices_before_max[0]
 116        if significant_indices_before_max.size > 0 else idx_max)
 117
 118  significant_indices_from_max = np.where(
 119      abs_array[idx_max:] >= threshold * array_max_val)[0]
 120  i2 = (idx_max + significant_indices_from_max[-1] + 1
 121        if significant_indices_from_max.size > 0 else idx_max + 1)
 122  return i1, i2

 125def get_significant_part_indices_v2(
 126    array_data: ArrayLike,
 127    threshold: float = 1e-8) -> Tuple[int, int]:
 128  """Returns indices (i1, i2) based on parts of the array that are "small".
 129
 130  The interpretation of (i1, i2) from the original code is:
 131  `i1` is the index of the last element *before* the peak region that is
 132  considered small (abs(element) < threshold * max_val).
 133  `i2` is the index *after* the first element *after* the peak region that
 134  is considered small.
 135  The docstring of the original function was "Return a tuple (i1,i2) such
 136  that none of the elements A[i1:i2] is small", which might be misleading
 137  as A[i1] and A[i2-1] could themselves be small by this definition.
 138  A slice like A[i1+1 : i2-1] or A[i1+1 : idx_first_small_after_peak]
 139  might better correspond to "all elements are not small".
 140
 141  Args:
 142    array_data: The input 1D array.
 143    threshold: The relative threshold to determine smallness.
 144               Elements are small if abs(element) < threshold * max(abs(array_data)).
 145
 146  Returns:
 147    A tuple (i1, i2).
 148  """
 149  abs_array = np.abs(array_data)
 150  if abs_array.size == 0:
 151    return 0, 0
 152
 153  idx_max = np.argmax(abs_array)
 154  array_max_val = abs_array[idx_max]
 155
 156  if array_max_val == 0: # All elements are zero
 157      return 0, len(array_data)
 158
 159  small_indices_before_max = np.where(
 160      abs_array[:idx_max] < threshold * array_max_val)[0]
 161  i1 = (small_indices_before_max[-1]
 162        if small_indices_before_max.size > 0 else 0)
 163
 164  small_indices_from_max = np.where(
 165      abs_array[idx_max:] < threshold * array_max_val)[0]
 166  # small_indices_from_max are relative to idx_max
 167  i2 = (idx_max + small_indices_from_max[0] + 1
 168        if small_indices_from_max.size > 0 else len(array_data))
 169  return i1, i2

 172def Fourier_filter(
 173    data: ArrayLike,
 174    time_step: float,
 175    spectral_window: ArrayLike,
 176    periodic: bool = False) -> ArrayLike:
 177  """Applies a Fourier filter to time-series data.
 178
 179  The function performs a Fourier transform, multiplies by a spectral
 180  window, and then performs an inverse Fourier transform.
 181
 182  Args:
 183    data: A 1D array or a 2D array (num_time_points, num_series)
 184          where each column is a time series.
 185    time_step: The time step between data points.
 186    spectral_window: A 2D array (num_window_points, 2) where the first
 187                     column contains circular frequencies (ascending) and
 188                     the second column contains the window function W(omega).
 189    periodic: If True, data is assumed to be periodic. Otherwise, data is
 190              mirrored and padded to reduce edge effects.
 191
 192  Returns:
 193    The filtered data, with the same shape as the input `data`.
 194  """
 195  if spectral_window.shape[0] == 0:
 196    return data.copy() # Return a copy to match behavior when filtering occurs
 197
 198  original_shape = data.shape
 199  if data.ndim == 1:
 200    # Reshape 1D array to 2D for consistent processing
 201    current_data = data.reshape(-1, 1)
 202  else:
 203    current_data = data.copy() # Work on a copy
 204
 205  num_time_points = current_data.shape[0]
 206
 207  if not periodic:
 208    # Mirror and concatenate data for non-periodic signals
 209    # Effectively doubles the number of time points for FFT
 210    current_data = np.vstack((current_data, current_data[::-1, :]))
 211    num_time_points_fft = current_data.shape[0]
 212  else:
 213    num_time_points_fft = num_time_points
 214
 215  # Fourier transform
 216  Fourier_transformed_data = np.fft.fftshift(
 217      np.fft.fft(current_data, axis=0), axes=0
 218  )
 219
 220  # Create frequency grid for the spectral window
 221  delta_omega = 2 * np.pi / (num_time_points_fft * time_step)
 222  # np.arange needs to create num_time_points_fft points
 223  # fftshift moves 0 frequency to the center.
 224  # The indices for fftshift range from -N/2 to N/2-1 (approx)
 225  omega_grid_indices = np.arange(num_time_points_fft) - np.floor(num_time_points_fft / 2.0)
 226  omega_grid = delta_omega * omega_grid_indices
 227  
 228  # Interpolate spectral window onto the data's frequency grid
 229  # Ensure omega_grid is 1D for interpolation if it became 2D due to reshape
 230  window_values = np.interp(
 231      np.abs(omega_grid.ravel()), # Use absolute frequencies
 232      spectral_window[:, 0],
 233      spectral_window[:, 1],
 234      left=1.0, # Value for frequencies below spectral_window range
 235      right=0.0 # Value for frequencies above spectral_window range
 236  )
 237  
 238  # Apply spectral filter
 239  # window_values needs to be (num_time_points_fft, 1) to broadcast
 240  Fourier_transformed_data *= window_values.reshape(-1, 1)
 241
 242  # Inverse Fourier transform
 243  filtered_data_full = np.fft.ifft(
 244      np.fft.ifftshift(Fourier_transformed_data, axes=0), axis=0
 245  )
 246
 247  if not periodic:
 248    # Truncate to original length if data was mirrored
 249    filtered_data_final = filtered_data_full[:num_time_points, :]
 250  else:
 251    filtered_data_final = filtered_data_full
 252
 253  # Ensure output is real if input was real
 254  if np.all(np.isreal(data)):
 255    filtered_data_final = filtered_data_final.real
 256
 257  return filtered_data_final.reshape(original_shape)

 260def polyfit_with_weights(x_coords: ArrayLike,
 261                         y_values: ArrayLike,
 262                         weights: ArrayLike,
 263                         degree: int) -> ArrayLike:
 264  """Performs a weighted least-squares polynomial fit.
 265
 266  Args:
 267    x_coords: 1D array of sample point x-coordinates.
 268    y_values: 1D array of sample point y-values to fit.
 269    weights: 1D array of weights for each sample point.
 270    degree: Degree of the fitting polynomial.
 271
 272  Returns:
 273    A 1D array of polynomial coefficients [p_degree, ..., p_1, p_0].
 274  """
 275  num_coeffs = degree + 1
 276  matrix_a = np.empty((num_coeffs, num_coeffs), dtype=np.float64)
 277  weights_squared = weights * weights
 278
 279  # Construct the Vandermonde-like matrix A for the normal equations
 280  # A[i,j] = sum(w_k^2 * x_k^(i+j))
 281  for i in range(num_coeffs):
 282    for j in range(i, num_coeffs):
 283      matrix_a[i, j] = np.sum(weights_squared * (x_coords**(i + j)))
 284      if i != j:
 285        matrix_a[j, i] = matrix_a[i, j] # Symmetric matrix
 286
 287  # Construct the vector b for the normal equations
 288  # b[i] = sum(w_k^2 * y_k * x_k^i)
 289  vector_b = np.empty(num_coeffs, dtype=np.float64)
 290  for i in range(num_coeffs):
 291    vector_b[i] = np.sum(weights_squared * y_values * (x_coords**i))
 292
 293  # Solve the linear system A * p = b for coefficients p
 294  solution_coeffs = scipy.linalg.solve(matrix_a, vector_b)
 295  return solution_coeffs[::-1] # Return in conventional order (highest power first)

 298def Fourier_transform(
 299    time_points: np.ndarray,
 300    y_data: np.ndarray,
 301    target_frequencies: Optional[np.ndarray] = None,
 302    is_periodic: bool = False,
 303    pulse_center_times: Optional[Union[float, np.ndarray]] = None
 304) -> Union[np.ndarray, Tuple[np.ndarray, np.ndarray]]:
 305    r"""Applies the Fast Fourier Transform (FFT) in an easy-to-use way.
 306
 307    This function computes the Fourier transform of time-dependent data Y(t),
 308    defined as:
 309        $F[Y](\omega) = \int_{-\infty}^\infty dt Y(t) \exp(i \omega t)$
 310
 311    Args:
 312        time_points: A 1D NumPy array of shape (N_t,) representing the time
 313            discretization of Y(t). Must be sorted in ascending order. N_t must
 314            be greater than 1.
 315        y_data: A NumPy array of shape (N_t, N_data) or (N_t,) containing the
 316            time-dependent data to Fourier transform. If 1D, it's treated as
 317            a single data series.
 318        target_frequencies: An optional 1D NumPy array of shape (N_omega,)
 319            specifying the circular frequencies at which to compute the
 320            transform. Must be sorted in ascending order. If None, the transform
 321            is returned on an optimal internal frequency grid.
 322        is_periodic: A boolean indicating if Y(t) is assumed to be periodic.
 323            If True, the time step is expected to be constant, and
 324            Y(t_end + dt) = Y(t_start). If False, precautions are taken to
 325            minimize artifacts from the DFT's implicit periodicity assumption.
 326        pulse_center_times: An optional scalar or 1D NumPy array of length
 327            N_data. If the input data represents a pulse not centered in its
 328            time window, specifying the center of each pulse (t0) helps avoid
 329            interpolation artifacts. If None, the center of the time window is
 330            used.
 331
 332    Returns:
 333        If `target_frequencies` is None:
 334            A tuple `(transformed_y, fft_omega_grid)`, where:
 335            - `transformed_y` (np.ndarray): The Fourier-transformed data,
 336              shape (N_fft, N_data) or (N_fft,) if input `y_data` was 1D.
 337            - `fft_omega_grid` (np.ndarray): The array of circular frequencies
 338              (shape (N_fft,)) corresponding to `transformed_y`.
 339        If `target_frequencies` is not None:
 340            An np.ndarray of shape (N_omega, N_data) or (N_omega,) containing
 341            the Fourier transform of `y_data` evaluated at the specified
 342            `target_frequencies`.
 343
 344    Raises:
 345        ValueError: If `time_points` has fewer than 2 elements.
 346    """
 347    _MAX_FFT_POINTS = 2**20
 348    num_time_points = len(time_points)
 349    if num_time_points <= 1:
 350        raise ValueError(
 351            "Input 'time_points' array must have more than one element."
 352        )
 353
 354    original_y_ndim = y_data.ndim
 355    if original_y_ndim == 1:
 356        # Reshape 1D y_data to (N_t, 1) for consistent processing
 357        y_data_processed = y_data.reshape((num_time_points, 1))
 358    else:
 359        y_data_processed = y_data
 360
 361    num_data_series = y_data_processed.shape[1]
 362
 363    # Determine the minimum time step from the input time_points
 364    min_time_step = np.min(np.diff(time_points))
 365    # This will be the effective time step for FFT. It might be modified later
 366    # for non-periodic cases to meet Nyquist criteria for target_frequencies.
 367    effective_time_step = min_time_step
 368
 369    period_duration: Optional[float] = None
 370    if is_periodic:
 371        # For periodic signals, the period is the total duration assuming
 372        # a constant step.
 373        period_duration = (time_points[-1] - time_points[0] +
 374                           np.mean(np.diff(time_points)))
 375
 376    # Handle the case where only one target frequency is given (no FFT needed)
 377    if target_frequencies is not None and len(target_frequencies) == 1:
 378        # Direct integration using trapezoidal rule
 379        # integrand shape: (num_time_points, num_data_series)
 380        integrand = y_data_processed * np.exp(
 381            1j * target_frequencies * time_points.reshape((num_time_points, 1))
 382        )
 383        # result shape: (num_data_series,)
 384        output_spectrum = np.trapezoid(integrand, time_points, axis=0)
 385        if is_periodic:
 386            # Correction for periodic functions with trapezoidal rule
 387            output_spectrum += 0.5 * min_time_step * (
 388                integrand[0, :] + integrand[-1, :]
 389            )
 390        # Reshape to (1, num_data_series) to match expected output shape
 391        output_spectrum = output_spectrum.reshape(1, num_data_series)
 392        if original_y_ndim == 1:
 393            return output_spectrum.reshape(len(target_frequencies)) # or .flatten()
 394        return output_spectrum
 395
 396    # Determine the target frequency resolution for FFT grid calculation
 397    dw_target_for_n_fft_calc: float
 398    if target_frequencies is None:
 399        # If no target frequencies, FFT resolution is based on input time window
 400        dw_target_for_n_fft_calc = (2 * np.pi /
 401                                   (min_time_step * num_time_points))
 402    else:
 403        # If target frequencies are given, use their minimum spacing
 404        dw_target_for_n_fft_calc = np.min(np.diff(target_frequencies))
 405
 406    # Determine the number of points for FFT (num_fft_points)
 407    num_fft_points: int
 408    if is_periodic:
 409        num_fft_points = num_time_points
 410        # If target frequencies are specified, may need to upsample N_fft
 411        # to achieve finer resolution than the default N_t points.
 412        # effective_time_step remains min_time_step for periodic case.
 413        if target_frequencies is not None:
 414            while (2 * np.pi / (effective_time_step * num_fft_points) >
 415                   1.1 * dw_target_for_n_fft_calc and
 416                   num_fft_points < _MAX_FFT_POINTS):
 417                num_fft_points *= 2
 418    else: # Not periodic
 419        # Initial estimate for num_fft_points based on desired resolution
 420        num_fft_points = 2**int(
 421            round(
 422                np.log(
 423                    2 * np.pi / (min_time_step * dw_target_for_n_fft_calc)
 424                ) / np.log(2.0)
 425            )
 426        )
 427        # Ensure FFT time window is large enough to cover the original data span
 428        while ((num_fft_points - 1) * min_time_step <
 429               time_points[-1] - time_points[0] and
 430               num_fft_points < _MAX_FFT_POINTS):
 431            num_fft_points *= 2
 432
 433        # For non-periodic signals with specified target_frequencies,
 434        # adjust effective_time_step and num_fft_points to satisfy Nyquist
 435        # and resolution requirements for the target_frequencies.
 436        if target_frequencies is not None:
 437            # Use temporary variables for this iterative adjustment
 438            current_dt = min_time_step
 439            current_n_fft = num_fft_points
 440            while (current_n_fft < _MAX_FFT_POINTS and
 441                   (target_frequencies[-1] >
 442                    np.pi / current_dt * (1.0 - 2.0 / current_n_fft) or
 443                    -target_frequencies[0] > np.pi / current_dt)):
 444                current_dt /= 2.0
 445                current_n_fft *= 2
 446            effective_time_step = current_dt
 447            num_fft_points = current_n_fft
 448
 449    # FFT time grid generation
 450    # This grid is centered around 0 after shifting by pulse_center_times
 451    fft_time_grid = effective_time_step * np.fft.ifftshift(
 452        np.arange(num_fft_points) - num_fft_points // 2
 453    )
 454
 455    # Determine effective pulse center times for interpolation and phase shift
 456    # pulse_centers_for_interp: used to shift time_points before interpolation
 457    # pulse_centers_for_phase: used for final phase correction, shape (1, N_data)
 458    
 459    _pulse_centers_for_interp: Union[float, np.ndarray]
 460    _pulse_centers_for_phase: np.ndarray
 461
 462    if pulse_center_times is None:
 463        # If no t0 provided, use the center of the time window
 464        time_window_center = 0.5 * (time_points[0] + time_points[-1])
 465        # Find the closest point in time_points to this center
 466        idx_center = np.argmin(np.abs(time_points - time_window_center))
 467        # This becomes the t0 for all data series
 468        calculated_t0 = time_points[idx_center]
 469        _pulse_centers_for_interp = calculated_t0 # Scalar
 470        _pulse_centers_for_phase = calculated_t0 * np.ones(
 471            (1, num_data_series)
 472        )
 473    else:
 474        _pulse_centers_for_interp = pulse_center_times # Can be scalar or array
 475        if np.isscalar(pulse_center_times):
 476            _pulse_centers_for_phase = pulse_center_times * np.ones(
 477                (1, num_data_series)
 478            )
 479        else:
 480            # Ensure it's a 1D array before reshaping
 481            _pulse_centers_for_phase = np.asarray(pulse_center_times).reshape(
 482                (1, num_data_series)
 483            )
 484
 485
 486    # Interpolate y_data onto the FFT time grid
 487    # y_interpolated_on_fft_grid shape: (num_fft_points, num_data_series)
 488    y_interpolated_on_fft_grid = np.zeros(
 489        (num_fft_points, num_data_series), dtype=y_data_processed.dtype
 490    )
 491
 492    for j_col in range(num_data_series):
 493        current_t0_interp: float
 494        if np.isscalar(_pulse_centers_for_interp):
 495            current_t0_interp = float(_pulse_centers_for_interp)
 496        else:
 497            current_t0_interp = float(np.asarray(_pulse_centers_for_interp)[j_col])
 498        
 499        # Shift original time_points relative to the current pulse center
 500        shifted_time_points = time_points - current_t0_interp
 501
 502        if is_periodic:
 503            # For periodic data, use period in interpolation
 504            y_interpolated_on_fft_grid[:, j_col] = np.interp(
 505                fft_time_grid,
 506                shifted_time_points,
 507                y_data_processed[:, j_col],
 508                period=period_duration
 509            )
 510        else:
 511            # For non-periodic, pad with zeros outside original time range
 512            y_interpolated_on_fft_grid[:, j_col] = np.interp(
 513                fft_time_grid,
 514                shifted_time_points,
 515                y_data_processed[:, j_col],
 516                left=0.0,
 517                right=0.0
 518            )
 519
 520    # Perform FFT
 521    # The result of ifft is scaled by (1/N). We multiply by (N*dt) to approximate
 522    # the integral definition F(omega) = integral Y(t)exp(i*omega*t) dt.
 523    # So, overall scaling is dt.
 524    y_fft = np.fft.fftshift(
 525        np.fft.ifft(y_interpolated_on_fft_grid, axis=0), axes=0
 526    ) * (num_fft_points * effective_time_step)
 527
 528    # FFT omega grid
 529    fft_omega_grid_spacing = (2 * np.pi /
 530                              (effective_time_step * num_fft_points))
 531    fft_omega_grid = fft_omega_grid_spacing * (
 532        np.arange(num_fft_points) - num_fft_points // 2
 533    )
 534
 535    # Apply phase correction due to pulse_center_times (t0)
 536    # This accounts for the shift Y(t) -> Y(t-t0) in time domain,
 537    # which corresponds to F(omega) -> F(omega) * exp(i*omega*t0)
 538    # if the FFT was performed on data effectively centered at t'=0.
 539    # The interpolation shifted data by -t0, so Z was Y(t').
 540    # The FFT of Y(t') is F[Y(t')] = integral Y(t')exp(iwt')dt'.
 541    # We want F[Y(t)] = integral Y(t)exp(iwt)dt.
 542    # F[Y(t)] = exp(iw t0_effective) * F[Y(t')].
 543    phase_correction = np.exp(
 544        1j * _pulse_centers_for_phase * fft_omega_grid.reshape((num_fft_points, 1))
 545    )
 546    y_fft_corrected = y_fft * phase_correction
 547
 548
 549    if target_frequencies is None:
 550        # Return FFT result on its own grid
 551        if original_y_ndim == 1:
 552            return y_fft_corrected.flatten(), fft_omega_grid
 553        return y_fft_corrected, fft_omega_grid
 554    else:
 555        # Interpolate FFT result onto the target_frequencies grid
 556        output_spectrum = np.zeros(
 557            (len(target_frequencies), num_data_series), dtype=np.complex128
 558        )
 559        for j_col in range(num_data_series):
 560            # Note: y_fft_corrected already includes the phase shift based on _pulse_centers_for_phase
 561            # and fft_omega_grid. When interpolating to target_frequencies, this phase is implicitly
 562            # interpolated as well.
 563             output_spectrum[:, j_col] = np.interp(
 564                target_frequencies,
 565                fft_omega_grid,
 566                y_fft_corrected[:, j_col], # Use the phase-corrected FFT result
 567                left=0.0,
 568                right=0.0
 569            )
 570        
 571        # The phase correction was already applied to y_fft before interpolation.
 572        # If we were to apply it *after* interpolation, it would be:
 573        # phase_correction_on_target_freq = np.exp(
 574        #    1j * _pulse_centers_for_phase * target_frequencies.reshape((len(target_frequencies), 1))
 575        # )
 576        # output_spectrum = interpolated_unphased_result * phase_correction_on_target_freq
 577        # However, the original code applies the phase correction *before* this final interpolation step
 578        # if omega is None, and *after* if omega is not None.
 579        # Let's re-check original logic for omega not None:
 580        # Z = np.fft.fftshift(np.fft.ifft(Z, axis=0), axes=0) * (N_fft * dt) <-- y_fft (unphased by t0 yet for this path)
 581        # ...
 582        # result[:,j] = np.interp(omega, w_grid, Z[:,j], left=0.0, right=0.0) <-- interpolation of unphased
 583        # result = result * np.exp(1j * t0 * omega.reshape((len(omega), 1))) <-- phase correction
 584        # This means my current y_fft_corrected (which has phase) should NOT be used for interpolation here.
 585        # I should interpolate 'y_fft' (before t0 correction) and then apply t0 correction using target_frequencies.
 586
 587        # Reverting to match original logic for target_frequencies path:
 588        # Interpolate the raw FFT result (before t0 correction)
 589        interpolated_raw_fft = np.zeros(
 590            (len(target_frequencies), num_data_series), dtype=np.complex128
 591        )
 592        for j_col in range(num_data_series):
 593            interpolated_raw_fft[:, j_col] = np.interp(
 594                target_frequencies,
 595                fft_omega_grid,
 596                y_fft[:, j_col], # Use y_fft (before _pulse_centers_for_phase correction)
 597                left=0.0,
 598                right=0.0
 599            )
 600        
 601        # Now apply phase correction using _pulse_centers_for_phase and target_frequencies
 602        phase_correction_final = np.exp(
 603            1j * _pulse_centers_for_phase * target_frequencies.reshape((len(target_frequencies), 1))
 604        )
 605        output_spectrum = interpolated_raw_fft * phase_correction_final
 606
 607        if original_y_ndim == 1:
 608            return output_spectrum.flatten()
 609        return output_spectrum

 612def inverse_Fourier_transform(
 613    omega_points: ArrayLike,
 614    data_series: ArrayLike,
 615    time_points_target: Optional[ArrayLike] = None,
 616    is_periodic: bool = False,
 617    frequency_offset: Optional[Union[float, ArrayLike]] = None
 618) -> Union[Tuple[ArrayLike, ArrayLike], ArrayLike]:
 619  r"""Applies inverse FFT to frequency-dependent data.
 620
 621  Computes $ F^{-1}[Y](t) = 1 / (2 \pi) \int_{-\infty}^\infty d\omega Y(\omega) \exp(-i t \omega) $.
 622
 623  Args:
 624    omega_points: 1D array of circular frequencies (N_omega), sorted.
 625    data_series: Frequency-dependent data (N_omega) or (N_omega, N_data_series).
 626    time_points_target: Optional 1D array of time points (N_t), sorted.
 627                        If None, times are determined by IFFT.
 628    is_periodic: If True, data_series is assumed periodic in frequency.
 629    frequency_offset: Scalar or array (N_data_series). Central frequency
 630                      offset(s) if data is not centered at omega=0.
 631
 632  Returns:
 633    If time_points_target is None:
 634      A tuple (transformed_data, time_grid).
 635    If time_points_target is provided:
 636      Transformed data interpolated at the given time_points_target.
 637  """
 638  # IFFT(Y(w)) = 1/(2pi) FT(Y(w))_at_-t = 1/(2pi) conj(FT(conj(Y(w)))_at_t)
 639  # The provided Fourier_transform computes FT[Y(t)](omega) = integral Y(t) exp(iwt) dt
 640  # We want IFT[Y(w)](t) = 1/(2pi) integral Y(w) exp(-iwt) dw
 641  # Let w' = -w, dw' = -dw.
 642  # = -1/(2pi) integral Y(-w') exp(iw't) dw' (from -inf to +inf, so limits flip)
 643  # = 1/(2pi) integral Y(-w') exp(iw't) dw' (from -inf to +inf)
 644  # So, call Fourier_transform with omega -> -omega (reversed), Y(omega) -> Y(-omega) (reversed)
 645  # and then scale by 1/(2pi). The 't' in Fourier_transform becomes our 'omega_points',
 646  # and 'omega' in Fourier_transform becomes our '-time_points_target'.
 647
 648  if time_points_target is None:
 649    # Transform Y(omega) as if it's a time signal Y(t=omega)
 650    # The 'omega' output of Fourier_transform will correspond to '-t'
 651    transformed_data, neg_time_grid = Fourier_transform(
 652        time_points=omega_points,
 653        y_data=data_series,
 654        target_frequencies=None, # Let FT determine output grid
 655        is_periodic=is_periodic,
 656        pulse_center_times=frequency_offset # This is omega0, an offset in the input "time" (omega) domain
 657    )
 658    # Result is FT[Y](k), where k is frequency. Here k corresponds to -t.
 659    # So, FT[Y(omega)](-t). We need to flip t and scale.
 660    return transformed_data[::-1] / (2 * np.pi), -neg_time_grid[::-1]
 661  else:
 662    # Target 'omega' for Fourier_transform is -time_points_target
 663    neg_target_times = -time_points_target[::-1] # Ensure it's sorted for FT
 664    
 665    result_at_neg_t = Fourier_transform(
 666        time_points=omega_points,
 667        y_data=data_series,
 668        target_frequencies=neg_target_times,
 669        is_periodic=is_periodic,
 670        pulse_center_times=frequency_offset
 671    )
 672    # result_at_neg_t is FT[Y(omega)](-t_target_sorted)
 673    # We want values at t_target, so reverse the order back.
 674    return result_at_neg_t[::-1] / (2 * np.pi)

 677def find_zero_crossings(x_values: ArrayLike,
 678                        y_values: ArrayLike) -> ArrayLike:
 679    """Finds all x-values where linearly interpolated y(x) = 0.
 680    
 681    Args:
 682      x_values: 1D array of x-coordinates, sorted ascending, no duplicates.
 683      y_values: 1D array of y-coordinates, same shape as x_values.
 684    
 685    Returns:
 686      A 1D array of x-values where y(x) crosses zero. Empty if no crossings.
 687    """
 688    if x_values.size == 0 or y_values.size == 0:
 689        return np.array([])
 690    if x_values.size != y_values.size:
 691        raise ValueError("x_values and y_values must have the same length.")
 692        
 693    # Product of y[i] and y[i+1]
 694    product_adjacent_y = y_values[:-1] * y_values[1:]
 695    crossings_x_coords: List[float] = []
 696      
 697    # Find indices where product is <= 0 (indicates a zero crossing or y[i]=0)
 698    for i in np.where(product_adjacent_y <= 0)[0]:
 699        # Instead of: if product_adjacent_y[i] == 0:
 700        #                 if y_values[i] == 0:
 701        # Use np.isclose for checking if y_values[i] or y_values[i+1] are zero
 702        y1_is_zero = np.isclose(y_values[i], 0.0)
 703        y2_is_zero = np.isclose(y_values[i+1], 0.0)
 704    
 705        if y1_is_zero and y2_is_zero: # segment is [0,0]
 706            crossings_x_coords.append(x_values[i])
 707            # To avoid double adding x_values[i+1] if it's processed as y1_is_zero in next iter
 708        elif y1_is_zero:
 709            crossings_x_coords.append(x_values[i])
 710        elif y2_is_zero and product_adjacent_y[i] < 0 : # Crosses and lands on zero at y2
 711             # The interpolation formula will give x_values[i+1]
 712             x1, x2_pt = x_values[i], x_values[i+1]
 713             y1_pt, y2_pt = y_values[i], y_values[i+1] # y2_pt is close to 0
 714             crossings_x_coords.append((x1*y2_pt - x2_pt*y1_pt) / (y2_pt - y1_pt))
 715        elif product_adjacent_y[i] < 0: # Definite crossing, neither is zero
 716            x1, x2 = x_values[i], x_values[i+1]
 717            y1_val, y2_val = y_values[i], y_values[i+1]
 718            crossings_x_coords.append((x1 * y2_val - x2 * y1_val) / (y2_val - y1_val))
 719    
 720    # Handle case where the last point itself is a zero not caught by pair product
 721    # This also needs np.isclose
 722    if y_values.size > 0 and np.isclose(y_values[-1], 0.0):
 723        # Avoid adding if it's already part of a segment ending in zero
 724        # that was captured by product_adjacent_y[i]=0 logic (where y[i+1]=0)
 725        already_found = False
 726        if crossings_x_coords and np.isclose(crossings_x_coords[-1], x_values[-1]):
 727            already_found = True
 728    
 729        if not already_found:
 730             # If y_values[-1] is zero, and y_values[-2]*y_values[-1] was not <=0 (e.g. y_values[-2] also zero)
 731             # or it was handled by interpolation which might be slightly off.
 732             # We want to ensure grid points that are zero are included.
 733             # A simpler way: collect all interpolated, then add all x where y is zero, then unique.
 734             pass # The unique call later should handle it if x_values[-1] was added by main loop
 735    
 736    # A more robust approach for points exactly on the grid:
 737    # After interpolation, add all x_values where corresponding y_values are close to zero.
 738    if x_values.size > 0: # Ensure x_values is not empty
 739        grid_zeros = x_values[np.isclose(y_values, 0.0)]
 740        crossings_x_coords.extend(list(grid_zeros))
 741    
 742    return np.unique(np.array(crossings_x_coords))

 745def find_extrema_positions(x_values: ArrayLike,
 746                           y_values: ArrayLike) -> ArrayLike:
 747  """Finds x-positions of local extrema in y(x).
 748
 749  Extrema are found where the derivative y'(x) (approximated by finite
 750  differences) crosses zero.
 751
 752  Args:
 753    x_values: 1D array of x-coordinates, sorted ascending, no duplicates.
 754    y_values: 1D array of y-coordinates, same shape as x_values.
 755
 756  Returns:
 757    A 1D array of x-values where y(x) has local extrema. Empty if none.
 758  """
 759  if len(x_values) < 2 or len(y_values) < 2: # Need at least two points for a derivative
 760    return np.array([])
 761  if len(x_values) != len(y_values):
 762    raise ValueError("x_values and y_values must have the same length.")
 763
 764  # Approximate derivative y'(x)
 765  delta_y = y_values[1:] - y_values[:-1]
 766  delta_x = x_values[1:] - x_values[:-1]
 767  # Avoid division by zero if x_values have duplicates (though pre-condition says no duplicates)
 768  # However, if delta_x is extremely small, derivative can be huge.
 769  # For robustness, filter out zero delta_x if they somehow occur.
 770  valid_dx = delta_x != 0
 771  if not np.all(valid_dx): # Should not happen given preconditions
 772      delta_y = delta_y[valid_dx]
 773      delta_x = delta_x[valid_dx]
 774      mid_points_x_for_derivative = (x_values[1:] + x_values[:-1])[valid_dx] / 2.0
 775  else:
 776      mid_points_x_for_derivative = (x_values[1:] + x_values[:-1]) / 2.0
 777
 778  if delta_x.size == 0: # Not enough points after filtering
 779      return np.array([])
 780      
 781  derivative_y = delta_y / delta_x
 782  
 783  # Find where the derivative crosses zero
 784  extrema_x_coords = find_zero_crossings(mid_points_x_for_derivative,
 785                                         derivative_y)
 786  return extrema_x_coords

 789def minimize_imaginary_parts(complex_array: ArrayLike) -> ArrayLike:
 790  """Rotates a complex array by a phase to make it as close as possible to being real-valued
 791
 792  Multiplies `complex_array` by `exp(1j*phi)` where `phi` is chosen to
 793  minimize `sum(imag(exp(1j*phi) * complex_array)**2)`.
 794
 795  Args:
 796    complex_array: A NumPy array of complex numbers.
 797
 798  Returns:
 799    The phase-rotated complex NumPy array.
 800  """
 801  if complex_array.size == 0:
 802      return complex_array.copy()
 803      
 804  # Z = X + iY. We want to minimize sum( (X sin(phi) + Y cos(phi))^2 )
 805  # d/dphi (sum(...)) = 0 leads to tan(2*phi) = 2*sum(XY) / sum(Y^2 - X^2)
 806  real_part = complex_array.real
 807  imag_part = complex_array.imag
 808
 809  numerator = 2 * np.sum(real_part * imag_part)
 810  denominator = np.sum(imag_part**2 - real_part**2)
 811
 812  # arctan2 handles signs and denominator being zero correctly
 813  phi = 0.5 * np.arctan2(numerator, denominator)
 814
 815  # The arctan2 gives phi in (-pi, pi], so 0.5*phi is in (-pi/2, pi/2].
 816  # This finds one extremum. The other is phi + pi/2. We need the minimum.
 817  rotated_z1 = complex_array * np.exp(1j * phi)
 818  imag_energy1 = np.sum(rotated_z1.imag**2)
 819
 820  rotated_z2 = complex_array * np.exp(1j * (phi + 0.5 * np.pi))
 821  imag_energy2 = np.sum(rotated_z2.imag**2)
 822
 823  if imag_energy2 < imag_energy1:
 824    phi += 0.5 * np.pi
 825
 826  # Normalize phi to be in (-pi/2, pi/2] or a similar principal range if desired,
 827  # though for exp(1j*phi) it doesn't strictly matter beyond 2pi periodicity.
 828  # The original code maps phi to (-pi/2, pi/2] effectively.
 829  phi -= np.pi * np.round(phi / np.pi) # This maps to (-pi/2, pi/2]
 830  # Let's test the original normalization:
 831  # If phi = 0.6*pi, round(0.6) = 1. phi = 0.6pi - pi = -0.4pi. Correct.
 832  # If phi = 0.4*pi, round(0.4) = 0. phi = 0.4pi. Correct.
 833  # If phi = -0.6*pi, round(-0.6) = -1. phi = -0.6pi + pi = 0.4pi. Correct.
 834  # This normalization is fine.
 835
 836  return complex_array * np.exp(1j * phi)

 839def integrate_oscillating_function(
 840    x_values: ArrayLike,
 841    func_values: ArrayLike,
 842    phase_values: ArrayLike,
 843    phase_step_threshold: float = 1e-3
 844) -> ArrayLike:
 845  r"""Integrates f(x) * exp(i * phi(x)) for quickly oscillating functions.
 846
 847  Uses an algorithm suitable for integrating f(x) * exp(i * phi(x)) dx
 848  over small intervals, particularly when phi(x) changes rapidly.
 849
 850  Args:
 851    x_values: 1D array of sorted x-coordinates.
 852    func_values: Array of function values f(x). Can be 1D (N_x) or 2D
 853                 (N_x, N_series).
 854    phase_values: Array of real-valued phase phi(x). Same shape as func_values.
 855    phase_step_threshold: Small positive number. Prevents division by
 856                          small d_phi in the integration formula.
 857
 858  Returns:
 859    A scalar or 1D array (N_series) of integral results.
 860  """
 861  # Input validation
 862  if not (x_values.shape[0] == func_values.shape[0] == phase_values.shape[0]):
 863      raise ValueError("x_values, func_values, and phase_values must have "
 864                       "the same length along the integration axis (axis 0).")
 865  if not np.allclose(np.imag(phase_values), 0): # Ensure phase is real
 866      raise ValueError("phase_values must be real-valued.")
 867  if func_values.ndim > 1 and func_values.shape != phase_values.shape:
 868      raise ValueError("If func_values is 2D, phase_values must have the "
 869                       "exact same shape.")
 870
 871  delta_x = x_values[1:] - x_values[:-1]
 872  
 873  # Prepare f and phi for interval calculations
 874  f1 = func_values[:-1, ...] # f(x_i)
 875  f2 = func_values[1:, ...]  # f(x_{i+1})
 876  delta_f = f2 - f1
 877
 878  phi1 = phase_values[:-1, ...] # phi(x_i)
 879  phi2 = phase_values[1:, ...]  # phi(x_{i+1})
 880  delta_phi = phi2 - phi1
 881
 882  # Reshape delta_x to broadcast with f1, f2, etc.
 883  # If func_values is (N_x, N_series), delta_x needs to be (N_x-1, 1)
 884  reshape_dims = (-1,) + (1,) * (func_values.ndim - 1)
 885  delta_x_reshaped = delta_x.reshape(reshape_dims)
 886
 887  # Common factor for the integral segments
 888  common_factor_z = delta_x_reshaped * np.exp(0.5j * (phi1 + phi2))
 889  
 890  integral_segments = np.zeros_like(common_factor_z, dtype=complex)
 891
 892  # Mask for small phase changes (use simpler approximation)
 893  is_small_delta_phi = np.abs(delta_phi) < phase_step_threshold
 894
 895  # Case 1: Small delta_phi (dphi is small)
 896  if np.any(is_small_delta_phi):
 897    # Approximation: integral \approx dx * exp(i*phi_avg) * (f_avg + i/8 * dphi * df)
 898    # This seems to be a higher-order trapezoidal rule for oscillating functions.
 899    # Original: Z[s] = Z[s] * (0.5 * (f1[s] + f2[s]) + 0.125j * dphi[s] * df[s])
 900    # where Z[s] was common_factor_z[is_small_delta_phi]
 901    term_small_dphi = (0.5 * (f1[is_small_delta_phi] + f2[is_small_delta_phi]) +
 902                       0.125j * delta_phi[is_small_delta_phi] * delta_f[is_small_delta_phi])
 903    integral_segments[is_small_delta_phi] = common_factor_z[is_small_delta_phi] * term_small_dphi
 904
 905  # Case 2: Large delta_phi (use formula for oscillating part)
 906  is_large_delta_phi = ~is_small_delta_phi
 907  if np.any(is_large_delta_phi):
 908    # This is likely an approximation based on integration by parts or steepest descent.
 909    # Original: Z[s] = Z[s] / dphi[s]**2 * (exp_term * (df[s] - 1j*f2[s]*dphi[s]) -
 910    #                                     (df[s] - 1j*f1[s]*dphi[s]) / exp_term)
 911    # where Z[s] was common_factor_z[is_large_delta_phi] and exp_term = exp(0.5j * dphi[s])
 912    
 913    dphi_large = delta_phi[is_large_delta_phi]
 914    exp_half_j_dphi = np.exp(0.5j * dphi_large)
 915    
 916    term1 = exp_half_j_dphi * (delta_f[is_large_delta_phi] -
 917                               1j * f2[is_large_delta_phi] * dphi_large)
 918    term2 = (delta_f[is_large_delta_phi] -
 919             1j * f1[is_large_delta_phi] * dphi_large) / exp_half_j_dphi
 920             
 921    integral_segments[is_large_delta_phi] = (common_factor_z[is_large_delta_phi] / (dphi_large**2) *
 922                                        (term1 - term2))
 923
 924  return np.sum(integral_segments, axis=0)

 927def calculate_permittivity_from_delta_polarization(
 928    time_step: float,
 929    polarization_delta_response: ArrayLike, # P_delta
 930    omega_array: ArrayLike,
 931    momentum_relaxation_rate: float = 0.0,
 932    dephasing_time: Optional[float] = None,
 933    disregard_drift_current: bool = False,
 934    allow_for_linear_displacement: bool = True
 935) -> ArrayLike:
 936  r"""Evaluates permittivity from polarization induced by E(t) = delta(t).
 937
 938  Handles drift currents and coherent oscillations in the polarization response.
 939  The relationship is $\epsilon(\omega) = 1 + 4 \pi \chi(\omega)$, where
 940  $P(\omega) = \chi(\omega) E(\omega)$, and for $E(t)=\delta(t)$, $E(\omega)=1$.
 941  So $\chi(\omega) = P_{\delta}(\omega)$.
 942
 943  Args:
 944    time_step: Time step (atomic units) of the polarization grid.
 945    polarization_delta_response: 1D array of polarization response P(t)
 946                                 (atomic units) induced by E(t)=delta(t).
 947                                 P_delta[0] corresponds to t=time_step.
 948    omega_array: 1D array of circular frequencies (a.u.) for permittivity calculation.
 949                 All frequencies must be non-zero.
 950    momentum_relaxation_rate: If non-zero, models Drude-like momentum relaxation
 951                              (gamma in 1/(omega*(omega+i*gamma))).
 952    dephasing_time: If not None/zero, an exponential decay (rate 1/dephasing_time)
 953                    is applied to coherent dipole oscillations in P_delta.
 954    disregard_drift_current: If True, the J_drift component of polarization
 955                             has no effect on the result.
 956    allow_for_linear_displacement: If True, fits P(t) ~ J_drift*t + P_offset.
 957                                   If False, fits P(t) ~ J_drift*t (P_offset=0).
 958
 959  Returns:
 960    A complex array (same shape as omega_array) of permittivity values.
 961  """
 962  if not np.all(omega_array != 0):
 963      raise ValueError("All elements in omega_array must be non-zero.")
 964
 965  # Construct time grid and full polarization array P(t), P(0)=0
 966  num_p_delta_points = polarization_delta_response.size
 967  # P_delta starts at t=dt, so P has N_t = num_p_delta_points + 1 points
 968  num_time_points = num_p_delta_points + 1
 969  time_grid = time_step * np.arange(num_time_points)
 970  time_max = time_grid[-1]
 971
 972  polarization_full = np.zeros(num_time_points)
 973  polarization_full[1:] = polarization_delta_response # P(0)=0
 974
 975  # Fit and subtract linear trend (drift current and offset)
 976  # Fit is done on the latter half of the data
 977  fit_start_index = num_time_points // 2
 978  if fit_start_index < 2 and num_time_points >=2 : # Need at least 2 points for polyfit(deg=1)
 979      fit_start_index = 0 # Use all data if too short for half

1057    # and we assume P(t) = P_offset for t > t_max, its FT is P_offset * exp(i*omega*t_max) * (pi*delta(omega) + 1/(i*omega))

1059    # If the window makes processed_polarization(t_max) close to P_offset,
1060    # and we assume P(t) = P_offset for t > t_max, the integral from t_max to inf is
1061    # P_offset * integral_{t_max to inf} exp(i*omega*t) dt
1062    # = P_offset * [exp(i*omega*t) / (i*omega)]_{t_max to inf}
1063    # For convergence, Im(omega) > 0 or add damping. Assuming real omega, this diverges.
1064    # The term P_offset * 1j * np.exp(1j * omega * t_max) / omega
1065    # is -P_offset * exp(1j*omega*t_max) / (1j*omega). This is the upper limit of the integral.
1066    # It implies the lower limit (at infinity) is taken as zero.
1067    # This is the FT of P_offset * step_function(t_max - t) if integrated from -inf to t_max.
1068    # Or FT of P_offset for t > t_max, i.e. integral from t_max to infinity of P_offset*exp(iwt)

1073        chi_omega += P_offset * 1j * np.exp(1j * omega_val * time_max) / omega_val

1075    # Add contribution from drift current J_drift / (omega * (omega + i*gamma_relax))
1076    if not disregard_drift_current and not np.isclose(J_drift, 0.0):
1077      denominator_drift = omega_val * (omega_val + 1j * momentum_relaxation_rate)
1078      if not np.isclose(denominator_drift, 0.0): # Avoid division by zero if omega=0 (already checked) or omega=-i*gamma

1080      # If denominator is zero (e.g. omega_val=0 or omega_val = -1j*gamma), this term is singular.
1081      # omega_val != 0 is asserted. If omega_val = -1j*gamma, it's a resonant condition.

34def soft_window(x_grid: ArrayLike,
35                x_begin: float,
36                x_end: float) -> ArrayLike:
37  """Computes a soft window function.
38
39  The window smoothly transitions from 1 to 0 over the interval
40  [min(x_begin, x_end), max(x_begin, x_end)].
41  If x_begin <= x_end, it's a cosine-squared fall-off.
42  If x_begin > x_end, it's a sine-squared rise-on.
43
44  Args:
45    x_grid: A 1D array of x-coordinates.
46    x_begin: The x-coordinate where the transition starts (value is 1).
47    x_end: The x-coordinate where the transition ends (value is 0).
48
49  Returns:
50    A 1D array of the same size as x_grid, containing the window values.
51  """
52  window = np.zeros_like(x_grid)
53  x_min_transition = min(x_begin, x_end)
54  x_max_transition = max(x_begin, x_end)
55
56  # Determine indices for different parts of the window
57  indices_before_transition = np.where(x_grid < x_min_transition)[0]
58  idx_transition_start = (indices_before_transition[-1] + 1
59                          if indices_before_transition.size > 0 else 0)
60
61  indices_after_transition = np.where(x_grid > x_max_transition)[0]
62  idx_transition_end = (indices_after_transition[0]
63                        if indices_after_transition.size > 0
64                        else len(x_grid))
65
66  # Define the transition region
67  x_transition = x_grid[idx_transition_start:idx_transition_end]
68
69  if x_begin <= x_end: # Window goes from 1 down to 0
70    window[:idx_transition_start] = 1.0
71    if idx_transition_end > idx_transition_start and x_max_transition > x_min_transition:
72      window[idx_transition_start:idx_transition_end] = \
73          np.cos(np.pi / 2.0 * (x_transition - x_min_transition) /
74                 (x_max_transition - x_min_transition))**2
75    # Values after x_end remain 0 (initialized)
76  else: # Window goes from 0 up to 1 (x_begin > x_end, so x_min_transition = x_end)
77    window[idx_transition_end:] = 1.0
78    if idx_transition_end > idx_transition_start and x_max_transition > x_min_transition:
79      window[idx_transition_start:idx_transition_end] = \
80          np.sin(np.pi / 2.0 * (x_transition - x_min_transition) /
81                 (x_max_transition - x_min_transition))**2
82    # Values before x_begin (which is x_max_transition) remain 0 (initialized)
83  return window

335            A tuple `(transformed_y, fft_omega_grid)`, where:
336            - `transformed_y` (np.ndarray): The Fourier-transformed data,
337              shape (N_fft, N_data) or (N_fft,) if input `y_data` was 1D.
338            - `fft_omega_grid` (np.ndarray): The array of circular frequencies
339              (shape (N_fft,)) corresponding to `transformed_y`.
340        If `target_frequencies` is not None:
341            An np.ndarray of shape (N_omega, N_data) or (N_omega,) containing
342            the Fourier transform of `y_data` evaluated at the specified
343            `target_frequencies`.
344
345    Raises:
346        ValueError: If `time_points` has fewer than 2 elements.
347    """
348    _MAX_FFT_POINTS = 2**20
349    num_time_points = len(time_points)
350    if num_time_points <= 1:
351        raise ValueError(
352            "Input 'time_points' array must have more than one element."
353        )
354
355    original_y_ndim = y_data.ndim
356    if original_y_ndim == 1:
357        # Reshape 1D y_data to (N_t, 1) for consistent processing
358        y_data_processed = y_data.reshape((num_time_points, 1))
359    else:
360        y_data_processed = y_data
361
362    num_data_series = y_data_processed.shape[1]
363
364    # Determine the minimum time step from the input time_points
365    min_time_step = np.min(np.diff(time_points))
366    # This will be the effective time step for FFT. It might be modified later
367    # for non-periodic cases to meet Nyquist criteria for target_frequencies.
368    effective_time_step = min_time_step
369
370    period_duration: Optional[float] = None
371    if is_periodic:
372        # For periodic signals, the period is the total duration assuming
373        # a constant step.
374        period_duration = (time_points[-1] - time_points[0] +
375                           np.mean(np.diff(time_points)))
376
377    # Handle the case where only one target frequency is given (no FFT needed)
378    if target_frequencies is not None and len(target_frequencies) == 1:
379        # Direct integration using trapezoidal rule
380        # integrand shape: (num_time_points, num_data_series)
381        integrand = y_data_processed * np.exp(
382            1j * target_frequencies * time_points.reshape((num_time_points, 1))
383        )
384        # result shape: (num_data_series,)
385        output_spectrum = np.trapezoid(integrand, time_points, axis=0)
386        if is_periodic:
387            # Correction for periodic functions with trapezoidal rule
388            output_spectrum += 0.5 * min_time_step * (
389                integrand[0, :] + integrand[-1, :]
390            )
391        # Reshape to (1, num_data_series) to match expected output shape
392        output_spectrum = output_spectrum.reshape(1, num_data_series)
393        if original_y_ndim == 1:
394            return output_spectrum.reshape(len(target_frequencies)) # or .flatten()
395        return output_spectrum
396
397    # Determine the target frequency resolution for FFT grid calculation
398    dw_target_for_n_fft_calc: float
399    if target_frequencies is None:
400        # If no target frequencies, FFT resolution is based on input time window
401        dw_target_for_n_fft_calc = (2 * np.pi /
402                                   (min_time_step * num_time_points))
403    else:
404        # If target frequencies are given, use their minimum spacing
405        dw_target_for_n_fft_calc = np.min(np.diff(target_frequencies))
406
407    # Determine the number of points for FFT (num_fft_points)
408    num_fft_points: int
409    if is_periodic:
410        num_fft_points = num_time_points
411        # If target frequencies are specified, may need to upsample N_fft
412        # to achieve finer resolution than the default N_t points.
413        # effective_time_step remains min_time_step for periodic case.
414        if target_frequencies is not None:
415            while (2 * np.pi / (effective_time_step * num_fft_points) >
416                   1.1 * dw_target_for_n_fft_calc and
417                   num_fft_points < _MAX_FFT_POINTS):
418                num_fft_points *= 2
419    else: # Not periodic
420        # Initial estimate for num_fft_points based on desired resolution
421        num_fft_points = 2**int(
422            round(
423                np.log(
424                    2 * np.pi / (min_time_step * dw_target_for_n_fft_calc)
425                ) / np.log(2.0)
426            )
427        )
428        # Ensure FFT time window is large enough to cover the original data span
429        while ((num_fft_points - 1) * min_time_step <
430               time_points[-1] - time_points[0] and
431               num_fft_points < _MAX_FFT_POINTS):
432            num_fft_points *= 2
433
434        # For non-periodic signals with specified target_frequencies,
435        # adjust effective_time_step and num_fft_points to satisfy Nyquist
436        # and resolution requirements for the target_frequencies.
437        if target_frequencies is not None:
438            # Use temporary variables for this iterative adjustment
439            current_dt = min_time_step
440            current_n_fft = num_fft_points
441            while (current_n_fft < _MAX_FFT_POINTS and
442                   (target_frequencies[-1] >
443                    np.pi / current_dt * (1.0 - 2.0 / current_n_fft) or
444                    -target_frequencies[0] > np.pi / current_dt)):
445                current_dt /= 2.0
446                current_n_fft *= 2
447            effective_time_step = current_dt
448            num_fft_points = current_n_fft
449
450    # FFT time grid generation
451    # This grid is centered around 0 after shifting by pulse_center_times
452    fft_time_grid = effective_time_step * np.fft.ifftshift(
453        np.arange(num_fft_points) - num_fft_points // 2
454    )
455
456    # Determine effective pulse center times for interpolation and phase shift
457    # pulse_centers_for_interp: used to shift time_points before interpolation
458    # pulse_centers_for_phase: used for final phase correction, shape (1, N_data)
459    
460    _pulse_centers_for_interp: Union[float, np.ndarray]
461    _pulse_centers_for_phase: np.ndarray
462
463    if pulse_center_times is None:
464        # If no t0 provided, use the center of the time window
465        time_window_center = 0.5 * (time_points[0] + time_points[-1])
466        # Find the closest point in time_points to this center
467        idx_center = np.argmin(np.abs(time_points - time_window_center))
468        # This becomes the t0 for all data series
469        calculated_t0 = time_points[idx_center]
470        _pulse_centers_for_interp = calculated_t0 # Scalar
471        _pulse_centers_for_phase = calculated_t0 * np.ones(
472            (1, num_data_series)
473        )
474    else:
475        _pulse_centers_for_interp = pulse_center_times # Can be scalar or array
476        if np.isscalar(pulse_center_times):
477            _pulse_centers_for_phase = pulse_center_times * np.ones(
478                (1, num_data_series)
479            )
480        else:
481            # Ensure it's a 1D array before reshaping
482            _pulse_centers_for_phase = np.asarray(pulse_center_times).reshape(
483                (1, num_data_series)
484            )
485
486
487    # Interpolate y_data onto the FFT time grid
488    # y_interpolated_on_fft_grid shape: (num_fft_points, num_data_series)
489    y_interpolated_on_fft_grid = np.zeros(
490        (num_fft_points, num_data_series), dtype=y_data_processed.dtype
491    )
492
493    for j_col in range(num_data_series):
494        current_t0_interp: float
495        if np.isscalar(_pulse_centers_for_interp):
496            current_t0_interp = float(_pulse_centers_for_interp)
497        else:
498            current_t0_interp = float(np.asarray(_pulse_centers_for_interp)[j_col])
499        
500        # Shift original time_points relative to the current pulse center
501        shifted_time_points = time_points - current_t0_interp
502
503        if is_periodic:
504            # For periodic data, use period in interpolation
505            y_interpolated_on_fft_grid[:, j_col] = np.interp(
506                fft_time_grid,
507                shifted_time_points,
508                y_data_processed[:, j_col],
509                period=period_duration
510            )
511        else:
512            # For non-periodic, pad with zeros outside original time range
513            y_interpolated_on_fft_grid[:, j_col] = np.interp(
514                fft_time_grid,
515                shifted_time_points,
516                y_data_processed[:, j_col],
517                left=0.0,
518                right=0.0
519            )
520
521    # Perform FFT
522    # The result of ifft is scaled by (1/N). We multiply by (N*dt) to approximate
523    # the integral definition F(omega) = integral Y(t)exp(i*omega*t) dt.
524    # So, overall scaling is dt.
525    y_fft = np.fft.fftshift(
526        np.fft.ifft(y_interpolated_on_fft_grid, axis=0), axes=0
527    ) * (num_fft_points * effective_time_step)
528
529    # FFT omega grid
530    fft_omega_grid_spacing = (2 * np.pi /
531                              (effective_time_step * num_fft_points))
532    fft_omega_grid = fft_omega_grid_spacing * (
533        np.arange(num_fft_points) - num_fft_points // 2
534    )
535
536    # Apply phase correction due to pulse_center_times (t0)
537    # This accounts for the shift Y(t) -> Y(t-t0) in time domain,
538    # which corresponds to F(omega) -> F(omega) * exp(i*omega*t0)
539    # if the FFT was performed on data effectively centered at t'=0.
540    # The interpolation shifted data by -t0, so Z was Y(t').
541    # The FFT of Y(t') is F[Y(t')] = integral Y(t')exp(iwt')dt'.
542    # We want F[Y(t)] = integral Y(t)exp(iwt)dt.
543    # F[Y(t)] = exp(iw t0_effective) * F[Y(t')].
544    phase_correction = np.exp(
545        1j * _pulse_centers_for_phase * fft_omega_grid.reshape((num_fft_points, 1))
546    )
547    y_fft_corrected = y_fft * phase_correction
548
549
550    if target_frequencies is None:
551        # Return FFT result on its own grid
552        if original_y_ndim == 1:
553            return y_fft_corrected.flatten(), fft_omega_grid
554        return y_fft_corrected, fft_omega_grid
555    else:
556        # Interpolate FFT result onto the target_frequencies grid
557        output_spectrum = np.zeros(
558            (len(target_frequencies), num_data_series), dtype=np.complex128
559        )
560        for j_col in range(num_data_series):
561            # Note: y_fft_corrected already includes the phase shift based on _pulse_centers_for_phase
562            # and fft_omega_grid. When interpolating to target_frequencies, this phase is implicitly
563            # interpolated as well.
564             output_spectrum[:, j_col] = np.interp(
565                target_frequencies,
566                fft_omega_grid,
567                y_fft_corrected[:, j_col], # Use the phase-corrected FFT result
568                left=0.0,
569                right=0.0
570            )
571        
572        # The phase correction was already applied to y_fft before interpolation.
573        # If we were to apply it *after* interpolation, it would be:
574        # phase_correction_on_target_freq = np.exp(
575        #    1j * _pulse_centers_for_phase * target_frequencies.reshape((len(target_frequencies), 1))
576        # )
577        # output_spectrum = interpolated_unphased_result * phase_correction_on_target_freq
578        # However, the original code applies the phase correction *before* this final interpolation step
579        # if omega is None, and *after* if omega is not None.
580        # Let's re-check original logic for omega not None:
581        # Z = np.fft.fftshift(np.fft.ifft(Z, axis=0), axes=0) * (N_fft * dt) <-- y_fft (unphased by t0 yet for this path)
582        # ...
583        # result[:,j] = np.interp(omega, w_grid, Z[:,j], left=0.0, right=0.0) <-- interpolation of unphased
584        # result = result * np.exp(1j * t0 * omega.reshape((len(omega), 1))) <-- phase correction
585        # This means my current y_fft_corrected (which has phase) should NOT be used for interpolation here.
586        # I should interpolate 'y_fft' (before t0 correction) and then apply t0 correction using target_frequencies.
587
588        # Reverting to match original logic for target_frequencies path:
589        # Interpolate the raw FFT result (before t0 correction)
590        interpolated_raw_fft = np.zeros(
591            (len(target_frequencies), num_data_series), dtype=np.complex128
592        )
593        for j_col in range(num_data_series):
594            interpolated_raw_fft[:, j_col] = np.interp(
595                target_frequencies,
596                fft_omega_grid,
597                y_fft[:, j_col], # Use y_fft (before _pulse_centers_for_phase correction)
598                left=0.0,
599                right=0.0
600            )
601        
602        # Now apply phase correction using _pulse_centers_for_phase and target_frequencies
603        phase_correction_final = np.exp(
604            1j * _pulse_centers_for_phase * target_frequencies.reshape((len(target_frequencies), 1))
605        )
606        output_spectrum = interpolated_raw_fft * phase_correction_final
607
608        if original_y_ndim == 1:
609            return output_spectrum.flatten()
610        return output_spectrum

696    crossings_x_coords: List[float] = []
697      
698    # Find indices where product is <= 0 (indicates a zero crossing or y[i]=0)

738    # After interpolation, add all x_values where corresponding y_values are close to zero.

741        crossings_x_coords.extend(list(grid_zeros))

910    # Original: Z[s] = Z[s] / dphi[s]**2 * (exp_term * (df[s] - 1j*f2[s]*dphi[s]) -

923                                        (term1 - term2))
